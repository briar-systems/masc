# the x86 decode dispatch keys, in the order the dispatch prefers them: the opcode map and byte, then the code width,
# ModRM's mod and reg, and the prefixes an operand size, a rex2 prefix and a mandatory prefix read
#
#   key <name> | lo=<bit> len=<bits>     bits lo to lo + len - 1 of the match word
#
# the match word is masc's own, one word of what tells x86 rows apart, read under the code width through the prefix
# scan (masc.isa.x86.field.scan), so a byte is a prefix or an opcode as the width makes it. the generator states each
# row's cubes over these fields by their names, and masc's key module reads each field by the reader of its name,
# x86_key.<name>:
#   map:    the encoding space times four plus the opcode map
#   opcode: the opcode byte
#   mode:   the code width, 0 for 16, 1 for 32 and 2 for 64 bits
#   direct: 1 when ModRM.mod is 3, naming a register, else 0
#   reg:    ModRM.reg
#   osz:    1 when a 66 prefix is there
#   w:      the W bit of a rex or rex2 prefix
#   rex2:   1 when a rex2 prefix is there
#   rep:    the last of f2 and f3, 2 for f2 and 3 for f3, 0 for neither
# which keys the dispatch reads, and in what order, is masc's own choice, so no line overrides XED, which states the
# bits each pattern fixes and no key

key map | lo=0 len=4 | "Intel SDM 325462-093US Vol 2A 2.1.2 and 3.1.1.2, the escape bytes and their maps; XED datafiles/xed-isa.txt MAP" | -
key opcode | lo=4 len=8 | "Intel SDM 325462-093US Vol 2A 2.1.2, the opcode byte" | -
key mode | lo=12 len=2 | "XED datafiles/xed-state-bits.txt MODE, mode16 to mode64" | -
key direct | lo=14 len=1 | "Intel SDM 325462-093US Vol 2A 2.1.3, ModRM.mod 11 naming a register" | -
key reg | lo=15 len=3 | "Intel SDM 325462-093US Vol 2A 2.1.3, ModRM.reg as an opcode extension" | -
key osz | lo=18 len=1 | "Intel SDM 325462-093US Vol 2A 2.1.1, the operand-size override 66; XED datafiles/xed-state-bits.txt OSZ" | -
key w | lo=19 len=1 | "Intel SDM 325462-093US Vol 2A 2.2.1, rex.w; XED datafiles/xed-state-bits.txt REXW" | -
key rex2 | lo=20 len=1 | "Intel APX 355828-009US 3.1.2.1, the rex2 prefix; XED datafiles/xed-state-bits.txt REX2" | -
key rep | lo=22 len=2 | "Intel SDM 325462-093US Vol 2A 2.1.1, f2 and f3, the last of which holds" | -
