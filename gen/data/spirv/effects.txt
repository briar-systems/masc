# the control transfers of the spir-v rows, which the grammar does not state
#
#   control <opname> | transfer=<transfer> [returns=<yes|no>] target=<target> arg=<n>
#                                                    a control transfer, `conditional=yes` for one that may take
#                                                    another target than its operand's and `arg=<n>` for the operand
#                                                    holding its target. a branch with several targets names its first
#
# transfers are jump, call, return, trap and merge, targets direct and none. a trap states whether the
# invocation resumes after it (gen.control), and every one here ends the invocation or its stage, so none returns
control OpBranch | transfer=jump target=direct arg=0 | SPIR-V 1.6 rev 7, OpBranch | -
control OpBranchConditional | transfer=jump target=direct arg=1 conditional=yes | SPIR-V 1.6 rev 7, OpBranchConditional | -
control OpSwitch | transfer=jump target=direct arg=1 conditional=yes | SPIR-V 1.6 rev 7, OpSwitch | -
control OpReturn | transfer=return target=none | SPIR-V 1.6 rev 7, OpReturn | -
control OpReturnValue | transfer=return target=none | SPIR-V 1.6 rev 7, OpReturnValue | -
control OpFunctionCall | transfer=call target=direct arg=2 | SPIR-V 1.6 rev 7, OpFunctionCall | -
control OpKill | transfer=trap returns=no target=none | SPIR-V 1.6 rev 7, OpKill | -
control OpUnreachable | transfer=trap returns=no target=none | SPIR-V 1.6 rev 7, OpUnreachable | -
control OpTerminateInvocation | transfer=trap returns=no target=none | SPIR-V 1.6 rev 7, OpTerminateInvocation | -
control OpIgnoreIntersectionKHR | transfer=trap returns=no target=none | SPV_KHR_ray_tracing, OpIgnoreIntersectionKHR | -
control OpTerminateRayKHR | transfer=trap returns=no target=none | SPV_KHR_ray_tracing, OpTerminateRayKHR | -
control OpEmitMeshTasksEXT | transfer=trap returns=no target=none | SPV_EXT_mesh_shader, OpEmitMeshTasksEXT | -
control OpSelectionMerge | transfer=merge target=direct arg=0 | SPIR-V 1.6 rev 7, OpSelectionMerge | -
control OpLoopMerge | transfer=merge target=direct arg=0 | SPIR-V 1.6 rev 7, OpLoopMerge | -
