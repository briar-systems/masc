# the pseudo instructions of several instructions, which riscv-opcodes does not
# hold, read from the riscv-asm-manual's riscv-asm.md
#
#   operand <name> | immediate bits=<n> signed|unsigned fixup=<fixup> read   a symbol, an unscaled integer its fixup takes
#   operand <name> | register class=<class> write                            a register of the class the sequence builds
#   step <steps> <n> | <row> <parts>                                         instruction n of a run of steps, by its row's key
#   sequence <name> <n> | template=<text> operands=<operands> steps=<steps>  one sequence, `when=<mode>=<value>` and
#                                                                            `requires=<requirement>` where it holds alone
#
# a step's parts are its row's operands in order: `<n>` the sequence's operand
# n, `%<fixup>(<n>)` the symbol of operand n under the fixup, `%<fixup>(here)`
# the low part bound to the label of the high part, a number an immediate,
# and any other word a register by the register file's name. a fixup or a mode
# is masc's riscv fixup or mode named so, and a requirement is spelled as the
# files topic spells one. sequences of a name are numbered from 1, in the
# printer's preference, each name a constant of the first's index and a name
# of several also the count of them. the manual is not vendored, so nothing
# here overrides a source

operand target_call | immediate bits=32 signed fixup=call_plt_pair read | riscv-asm.md "Function Calls", the target the pair's relocation takes | -
operand rd | register class=x write | riscv-asm.md "Pseudoinstructions", the register built or linked | -
operand address | immediate bits=32 signed fixup=pcrel_hi20 read | riscv-asm.md "Pseudoinstructions", the symbol whose pc relative address lla builds | -
operand got | immediate bits=32 signed fixup=got_pcrel_hi20 read | riscv-asm.md "Pseudoinstructions", the symbol whose address la loads from the global offset table under pic | -

step call 1 | auipc ra %call_plt_pair(0) | riscv-asm.md "Function Calls", call through ra | -
step call 2 | jalr ra ra 0 | riscv-asm.md "Function Calls", call through ra | -
step call_linked 1 | auipc <0> %call_plt_pair(1) | riscv-asm.md "Function Calls", call through rd | -
step call_linked 2 | jalr <0> <0> 0 | riscv-asm.md "Function Calls", call through rd | -
step tail 1 | auipc t1 %call_plt_pair(0) | riscv-asm.md "Function Calls", tail through t1, linking nowhere | -
step tail 2 | jalr zero t1 0 | riscv-asm.md "Function Calls", tail through t1, linking nowhere | -
step lla 1 | auipc <0> %pcrel_hi20(1) | riscv-asm.md "Pseudoinstructions", lla | -
step lla 2 | addi <0> <0> %pcrel_lo12_i(here) | riscv-asm.md "Pseudoinstructions", lla | -
step got64 1 | auipc <0> %got_pcrel_hi20(1) | riscv-asm.md "Pseudoinstructions", la under pic, a load of the xlen | -
step got64 2 | ld <0> <0> %pcrel_lo12_i(here) | riscv-asm.md "Pseudoinstructions", la under pic, a load of the xlen | -
step got32 1 | auipc <0> %got_pcrel_hi20(1) | riscv-asm.md "Pseudoinstructions", la under pic, a load of the xlen | -
step got32 2 | lw <0> <0> %pcrel_lo12_i(here) | riscv-asm.md "Pseudoinstructions", la under pic, a load of the xlen | -

sequence call_linked 1 | template="call <0>, <1>" operands="rd target_call" steps=call_linked | riscv-asm.md "Function Calls" | -
sequence tail 1 | template="tail <0>" operands=target_call steps=tail | riscv-asm.md "Function Calls" | -
