# the control transfers of the wasm instructions, which opcode.def does not state
#
#   control <Name> | transfer=<jump|call|return|trap> target=<none|direct|indirect|depth> [arg=<n>] [conditional=yes]
#
# a branch's target is the label its operand counts outward to (effect.DEPTH), which the stream's label
# stack names, and br_table's every label and its default are its operand 0's list. a call's direct
# target is its function index, an indirect one the reference or table entry the operand stack holds,
# which no operand names. a tail call jumps. throw and throw_ref transfer to the handler the runtime
# finds, as a trap does. block, loop, if, else, try_table and end transfer by the structure the stream
# holds (structure.txt), so they state none here
#
# every line cites the WebAssembly Specification release 3.0, tag wg-3.0, 4.4.8 control instructions
# (specification/wasm-3.0/4.3-execution.instructions.spectec), as `exec`

control Unreachable | transfer=trap target=none | exec, unreachable traps | -
control Br | transfer=jump target=depth arg=0 | exec, br l jumps to the label l counts out to | -
control BrIf | transfer=jump target=depth arg=0 conditional=yes | exec, br_if l branches when its operand is not 0 | -
control BrTable | transfer=jump target=depth arg=0 | exec, br_table l* l branches to the label its operand indexes, or the default | -
control BrOnNull | transfer=jump target=depth arg=0 conditional=yes | exec, br_on_null l branches when the reference is null | -
control BrOnNonNull | transfer=jump target=depth arg=0 conditional=yes | exec, br_on_non_null l branches when the reference is not null | -
control Return | transfer=return target=none | exec, return | -
control Call | transfer=call target=direct arg=0 | exec, call x | -
control CallIndirect | transfer=call target=indirect | exec, call_indirect x y calls the table entry its operand indexes | -
control CallRef | transfer=call target=indirect | exec, call_ref x calls the reference on the stack | -
control ReturnCall | transfer=jump target=direct arg=0 | exec, return_call x | -
control ReturnCallIndirect | transfer=jump target=indirect | exec, return_call_indirect x y | -
control ReturnCallRef | transfer=jump target=indirect | exec, return_call_ref x | -
control Throw | transfer=trap target=none | exec, throw x transfers to the handler of its tag | -
control ThrowRef | transfer=trap target=none | exec, throw_ref transfers to the handler of the exception | -
