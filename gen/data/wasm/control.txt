# the control transfers of the wasm instructions, which opcode.def does not state
#
#   control <Name> | transfer=<jump|call|return|trap> [returns=<yes|no>] target=<none|direct|indirect|depth|structured>
#                    [arg=<n>] [conditional=yes] [condition=<n>] [chooses=<n>]
#
# a branch's target is the label its operand counts outward to (effect.DEPTH), which the stream's label
# stack names, and br_table's every label and its default are its operand 0's list. a call's direct
# target is its function index, an indirect one the reference or table entry the operand stack holds,
# which no operand names. a tail call jumps. throw and throw_ref transfer to the handler the runtime
# finds, as a trap does, and like unreachable never resume after it, so each is a trap that does not
# return (gen.control). if jumps to its second arm, or its end when it has none, when its condition
# is 0, and else from the end of the first arm to the end of the block, each where the structure it
# opens or splits goes on (effect.STRUCTURED, structure.txt). block, loop, try_table and end transfer
# nowhere, so they state none here. `condition` names the popped value that decides whether a
# conditional transfer is taken and `chooses` the popped value that chooses its target, each by its
# depth from the top of what the instruction pops (its signature, signatures.txt)
#
# every line cites the WebAssembly Specification release 3.0, tag wg-3.0, 4.4.8 control instructions
# (specification/wasm-3.0/4.3-execution.instructions.spectec), as `exec`

control Unreachable | transfer=trap returns=no target=none | exec, unreachable traps | -
control Br | transfer=jump target=depth arg=0 | exec, br l jumps to the label l counts out to | -
control BrIf | transfer=jump target=depth arg=0 conditional=yes condition=0 | exec, br_if l branches when its operand is not 0 | -
control BrTable | transfer=jump target=depth arg=0 chooses=0 | exec, br_table l* l branches to the label its operand indexes, or the default | -
control BrOnNull | transfer=jump target=depth arg=0 conditional=yes condition=0 | exec, br_on_null l branches when the reference is null | -
control BrOnNonNull | transfer=jump target=depth arg=0 conditional=yes condition=0 | exec, br_on_non_null l branches when the reference is not null | -
control If | transfer=jump target=structured conditional=yes condition=0 | exec, if bt instr1* else instr2* end runs instr2* when its operand is 0 | -
control Else | transfer=jump target=structured | exec, the label of if is its end, so the end of its first arm leaves the block | -
control Return | transfer=return target=none | exec, return | -
control Call | transfer=call target=direct arg=0 | exec, call x | -
control CallIndirect | transfer=call target=indirect chooses=0 | exec, call_indirect x y calls the table entry its operand indexes | -
control CallRef | transfer=call target=indirect chooses=0 | exec, call_ref x calls the reference on the stack | -
control ReturnCall | transfer=jump target=direct arg=0 | exec, return_call x | -
control ReturnCallIndirect | transfer=jump target=indirect chooses=0 | exec, return_call_indirect x y | -
control ReturnCallRef | transfer=jump target=indirect chooses=0 | exec, return_call_ref x | -
control Throw | transfer=trap returns=no target=none | exec, throw x transfers to the handler of its tag | -
control ThrowRef | transfer=trap returns=no target=none | exec, throw_ref transfers to the handler of the exception | -
