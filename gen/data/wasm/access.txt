# the memory, table and segment accesses and the index space entries of the wasm instructions, which
# opcode.def does not state
#
#   memarg [<prefix>] <lo>[-<hi>] | read|write|both
#   access <Name> | [read=<n>] [write=<n>] [both=<n>] [count=<n>] bits=<n>
#   entry <Name> | [read=<n>] [write=<n>] [both=<n>]
#   order [<prefix>] <lo>[-<hi>] | [before=<kinds>] [after=<kinds>]
#   ordered [<prefix>] <lo>[-<hi>] | atomic|fence
#
# an instruction with a memarg accesses the memory its memarg's index names, at the address it pops
# deepest, its t1 (signatures.txt), plus its offset, and as many bytes as opcode.def's memory column, so
# a memarg line states only whether the instructions of its range read the bytes, write them or both.
# every instruction with a memarg lies in exactly one range. an access line states each access of an
# instruction with no memarg, at the address it pops at that depth from the top of what it pops, over
# the count of elements it pops at `count`'s depth, one element when none, each element `bits` wide, 0
# for an element whose width the set does not state, as a table's reference. a data or element
# segment an instruction copies from is an access too. an entry line states each index space entry an
# instruction reads or writes through the operand of that number, a local or a global. an order line states how
# the instructions of its range order the accesses around them (gen.order), and an ordered line names a range whose
# every instruction must state an order
#
# every line cites the WebAssembly Specification release 3.0, tag wg-3.0, the execution of instructions
# (specification/wasm-3.0/4.3-execution.instructions.spectec), as `exec`, or the threads proposal's
# overview (proposals/threads/Overview.md), as `threads`

memarg 0x28-0x35 | read | exec, memory instructions, t.load and t.loadN_sx read their bytes from the address | -
memarg 0x36-0x3e | write | exec, memory instructions, t.store and t.storeN write their bytes at the address | -
memarg 0xfd 0x00-0x0a | read | exec, memory instructions, v128.load, v128.loadMxN_sx and v128.loadN_splat read | -
memarg 0xfd 0x0b | write | exec, memory instructions, v128.store writes | -
memarg 0xfd 0x54-0x57 | read | exec, memory instructions, v128.loadN_lane reads one lane | -
memarg 0xfd 0x58-0x5b | write | exec, memory instructions, v128.storeN_lane writes one lane | -
memarg 0xfd 0x5c-0x5d | read | exec, memory instructions, v128.loadN_zero reads | -
memarg 0xfe 0x00-0x02 | read | threads, wait and notify, memory.atomic.wait reads the expected value and notify addresses its waiters | -
memarg 0xfe 0x10-0x16 | read | threads, atomic memory accesses, the atomic loads | -
memarg 0xfe 0x17-0x1d | write | threads, atomic memory accesses, the atomic stores | -
memarg 0xfe 0x1e-0x4e | both | threads, read-modify-write instructions, each reads and writes its address | -

access MemoryCopy | write=2 read=1 count=0 bits=8 | exec, memory instructions, memory.copy x1 x2 copies n bytes from s to d | -
access MemoryFill | write=2 count=0 bits=8 | exec, memory instructions, memory.fill x writes n bytes from d | -
access MemoryInit | write=2 read=1 count=0 bits=8 | exec, memory instructions, memory.init x y copies n bytes of data segment y from s to d | -
access TableGet | read=0 bits=0 | exec, table instructions, table.get x reads the entry at i | -
access TableSet | write=1 bits=0 | exec, table instructions, table.set x writes the entry at i | -
access TableFill | write=2 count=0 bits=0 | exec, table instructions, table.fill x writes n entries from i | -
access TableCopy | write=2 read=1 count=0 bits=0 | exec, table instructions, table.copy x y copies n entries from s to d | -
access TableInit | write=2 read=1 count=0 bits=0 | exec, table instructions, table.init x y copies n entries of element segment y from s to d | -

entry LocalGet | read=0 | exec, variable instructions, local.get x reads local x | -
entry LocalSet | write=0 | exec, variable instructions, local.set x writes local x | -
entry LocalTee | write=0 | exec, variable instructions, local.tee x writes local x and keeps its value | -
entry GlobalGet | read=0 | exec, variable instructions, global.get x reads global x | -
entry GlobalSet | write=0 | exec, variable instructions, global.set x writes global x | -

order 0xfe 0x00-0x4e | before=rw after=rw | threads, the memory model, every atomic access, wait, notify and atomic.fence is sequentially consistent, so it orders every access before it against every access after it | -
ordered 0xfe 0x00-0x4e | atomic | threads, the atomic memory instructions and atomic.fence | -
