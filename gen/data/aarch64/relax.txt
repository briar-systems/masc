# the longer form of a branch whose target is out of range, and the code fill
#
#   relax <row> | to=<row> [flip=<mask> fixed=<member>,...]
#   far | row=<row>
#   fill | row=<row>
#
# a branch past its range becomes its inverse past the far branch to the target: tbz's fourteen bits reach 32KB and
# cbz's and b.cond's nineteen 1MB, b's twenty-six 128MB. a condition inverts by its low bit, but al and nv both mean
# always (DDI 0487 Table C1-1, "Standard condition codes"), so a branch on either has no inverse and no longer form.
# llvm-mc 23.1.1 and GNU as relax no aarch64 branch and refuse one out of range, so these are masc's own. the model
# states no relaxation, so no line overrides it

far | row=B_only_branch_imm | A64 ISA XML b_uncond.xml, "B": a branch to a label within 128MB | -
fill | row=NOP_HI_hints | A64 ISA XML nop.xml, "NOP"; llvm 23.1.1 AArch64AsmBackend::writeNopData writes 0xd503201f | -
relax BC_only_condbranch | to=BC_only_condbranch flip=1 fixed=al,nv | A64 ISA XML bc_cond.xml, "BC.cond": a label within 1MB; DDI 0487 Table C1-1, "Standard condition codes": al and nv are always | -
relax B_only_condbranch | to=B_only_condbranch flip=1 fixed=al,nv | A64 ISA XML b_cond.xml, "B.cond": a label within 1MB; DDI 0487 Table C1-1, "Standard condition codes": al and nv are always | -
relax CBNZ_32_compbranch | to=CBZ_32_compbranch | A64 ISA XML cbnz.xml, "CBNZ": a label within 1MB | -
relax CBNZ_64_compbranch | to=CBZ_64_compbranch | A64 ISA XML cbnz.xml, "CBNZ": a label within 1MB | -
relax CBZ_32_compbranch | to=CBNZ_32_compbranch | A64 ISA XML cbz.xml, "CBZ": a label within 1MB | -
relax CBZ_64_compbranch | to=CBNZ_64_compbranch | A64 ISA XML cbz.xml, "CBZ": a label within 1MB | -
relax TBNZ_only_testbranch.w | to=TBZ_only_testbranch.w | A64 ISA XML tbnz.xml, "TBNZ": a label within 32KB | -
relax TBNZ_only_testbranch.x | to=TBZ_only_testbranch.x | A64 ISA XML tbnz.xml, "TBNZ": a label within 32KB | -
relax TBZ_only_testbranch.w | to=TBNZ_only_testbranch.w | A64 ISA XML tbz.xml, "TBZ": a label within 32KB | -
relax TBZ_only_testbranch.x | to=TBNZ_only_testbranch.x | A64 ISA XML tbz.xml, "TBZ": a label within 32KB | -
